import FilterTable from '../../../../components/FilterTable.astro';
import { items, merchants, worldDrops, monsters, formatCoins, sellPrice, averageRoll } from '../../../../lib/gamedata.js';
import { itemName, itemDescriptions, slotNames, materialNames, categoryNames, monsterName } from '../../../../lib/en.js';

export const rica = merchants.find((m) => m.id === 'rica');
export const dropChance = new Map(worldDrops.map((d) => [d.id, d.chance]));
export const monsterWeapons = monsters.filter((m) => m.weapon);

export const sources = (id) => {
  const out = [];
  if (rica.catalog.includes(id)) out.push('Shop');
  if (dropChance.has(id)) out.push(`World drop ${(dropChance.get(id) * 100).toFixed(1)}%`);
  for (const m of monsterWeapons.filter((m) => m.weapon === id)) {
    out.push(`${monsterName(m.id)} ${Math.round(m.weaponDropChance * 100)}%`);
  }
  const it = items.find((i) => i.id === id);
  if (it?.chestTier) out.push(`Chest T${it.chestTier}${it.chestChance ? ` ${Math.round(it.chestChance * 100)}%` : ' guaranteed'}`);
  if (it?.grillsInto) out.push('Fishing');
  return out.length ? out.join(', ') : '—';
};

export const byId = (a, b) => a.id.localeCompare(b.id);

All **{items.length} items**, read straight from the game's `data-src/items.csv`
so they match what the server loads.

:::note[Reading prices]
The game stores one copper integer and converts on display:
**1 gold (g) = 100 silver (s) = 10,000 copper (c)**.
Items with no price (the "Worn" ones) are **refused by merchants** — that's the
design that stops starting gear being cashed out.

**Sell price** is what merchant Rica pays, fixed at
**{rica.sellRatePercent}%** of base. The spread means buying low and selling
high doesn't work.
:::

## Weapons

Damage is dice notation — `1d8` is one eight-sided die. The **average** column
is there so you can compare directly.

<FilterTable
  placeholder="Search weapons…"
  columns={[
    { key: 'name', label: 'Name' },
    { key: 'dice', label: 'Damage', align: 'center' },
    { key: 'avg', label: 'Avg', align: 'right' },
    { key: 'weight', label: 'Weight', align: 'right' },
    { key: 'price', label: 'Shop', align: 'right' },
    { key: 'sell', label: 'Sells for', align: 'right' },
    { key: 'source', label: 'Where to get it' },
  ]}
  rows={items
    .filter((i) => i.category === 'weapon')
    .sort((a, b) => (averageRoll(b.dice) ?? 0) - (averageRoll(a.dice) ?? 0))
    .map((i) => ({
      name: i.name,
      dice: i.dice ?? '—',
      avg: averageRoll(i.dice)?.toFixed(1) ?? '—',
      weight: `${i.weight} kg`,
      price: formatCoins(i.basePrice),
      sell: formatCoins(sellPrice(i.basePrice, rica.sellRatePercent)),
      source: sources(i.id),
    }))}
/>

## Armor

**Guard** is the number an attacker has to beat to hit you — higher is better.
Worn Guard adds up across slots.

<FilterTable
  placeholder="Search armor…"
  columns={[
    { key: 'name', label: 'Name' },
    { key: 'slot', label: 'Slot', align: 'center' },
    { key: 'guard', label: 'Guard', align: 'right' },
    { key: 'weight', label: 'Weight', align: 'right' },
    { key: 'material', label: 'Material', align: 'center' },
    { key: 'price', label: 'Shop', align: 'right' },
    { key: 'sell', label: 'Sells for', align: 'right' },
    { key: 'source', label: 'Where to get it' },
  ]}
  rows={items
    .filter((i) => i.category === 'armor')
    .sort((a, b) => (b.guard ?? 0) - (a.guard ?? 0))
    .map((i) => ({
      name: i.name,
      slot: slotNames[i.equipSlot] ?? i.equipSlot ?? '—',
      guard: i.guard ?? '—',
      weight: `${i.weight} kg`,
      material: materialNames[i.material] ?? i.material ?? '—',
      price: formatCoins(i.basePrice),
      sell: formatCoins(sellPrice(i.basePrice, rica.sellRatePercent)),
      source: sources(i.id),
    }))}
/>

The Gold Ring and Silver Necklace do nothing and **currently have no way to
obtain them** — they have no chest tier, so they're out of the pool. Every one
still in circulation dates from the old chest rules.

## Catches

Raw fish are edible (fixed 40 nutrition) but **carry a food poisoning risk**.
Grilling removes it. See [Fishing](../fishing/).

  placeholder="Search catches…"
  columns={[

    { key: 'weight', label: 'Weight', align: 'right' },
    { key: 'price', label: 'Shop', align: 'right' },
    { key: 'grill', label: 'Grills into' },

  rows={items
    .filter((i) => ['fish', 'junk', 'coin_catch'].includes(i.category))
    .sort((a, b) => (b.basePrice ?? 0) - (a.basePrice ?? 0))

      weight: `${i.weight} kg`,
      price: formatCoins(i.basePrice),
      grill: i.grillsInto ? itemName(i.grillsInto) : 'Cannot be grilled',
    }))}
/>
