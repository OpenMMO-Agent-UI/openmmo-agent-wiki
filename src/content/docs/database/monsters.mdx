import FilterTable from '../../../components/FilterTable.astro';
import { monsters, items, averageRoll, healthForLevel, attackBonusForLevel, damageRollForLevel } from '../../../lib/gamedata.js';
import { monsterName, itemName, behaviorNames } from '../../../lib/zh.js';

export const mark = (value, derived) => (derived ? `${value} *` : `${value}`);
export const byLevel = (a, b) => a.level - b.level;

目前共 **{monsters.length} 種**怪物。

:::caution[表中標了 `*` 的數值是推算出來的]
遊戲資料裡的生命、命中加值、傷害骰**留空代表「不覆寫」**,伺服器會依等級自動計算。
留空不等於 0,所以這裡照伺服器的公式(`server/src/game/combat.rs`)補上實際值,
並用 `*` 標示哪些是推算的:

- 生命 = `無條件進位(等級 × 9 ÷ 2)`
- 命中加值 = `等級`(怪物不像玩家那樣除以 2,見[戰鬥系統](../../guides/combat/))
- 傷害骰 = 依等級查表(見下方)
:::

## 全部怪物

<FilterTable
  placeholder="搜尋怪物…"

    .sort(byLevel)

      behavior: behaviorNames[m.behavior] ?? m.behavior,
    }))}
/>

**行為**只有兩種:*主動攻擊* 的怪物在視野內就會撲上來,*受攻擊才反擊* 的則要等你先動手才還手。
狗頭人和潛行者屬於後者,但兩種行為都會在玩家進入 **5 公尺**內時直接交戰——
「受攻擊才反擊」只代表不會遠距離主動撲過來,不代表可以貼身繞過。

玩家的基準移動速度是 **3.0**(吃飽有 ×1.1 加成,鎖定 3.3)。對照上表:
潛行者奔跑 8.0 遠比玩家快,**逃不掉**;獸人 4.5、哥布林酋長 4.0 也追得上你。
只有狗頭人(3.2)勉強能甩開。

## 出現位置

怪物分兩種出現方式:**地城**依樓層深度生成,**地表**則在指定區域生成。
權重越高代表在該深度範圍內出現得越頻繁。

<FilterTable
  placeholder="搜尋怪物…"
  columns={[
    { key: 'name', label: '名稱' },
    { key: 'depth', label: '地城深度' },
    { key: 'weight', label: '生成權重', align: 'right' },
    { key: 'note', label: '備註' },
  ]}
  rows={monsters
    .slice()
    .sort(byLevel)

    }))}
/>

## 掉落

怪物身上帶的武器就是它會掉的東西 —— 這是刻意的設計,**看得到就打得到**。
一般怪物掉落率 10%,頭目 100% 確定掉落。

<FilterTable
  placeholder="搜尋怪物或掉落物…"
  columns={[
    { key: 'name', label: '怪物' },
    { key: 'drop', label: '掉落武器' },
    { key: 'chance', label: '掉落率', align: 'right' },
    { key: 'dice', label: '武器傷害', align: 'center' },
  ]}
  rows={monsters
    .filter((m) => m.weapon)
    .sort(byLevel)
    .map((m) => ({
      name: monsterName(m.id),
      drop: itemName(m.weapon),
      chance: `${Math.round(m.weaponDropChance * 100)}%`,
      dice: items.find((i) => i.id === m.weapon)?.dice ?? '—',
    }))}
/>

獸人督軍是目前唯一確定掉落**鐵劍**(1d8)的來源 —— 鐵劍不在商店販售,
所以打倒它是拿到這把武器最直接的方法。

## 命中怎麼判定

攻擊時擲一顆 20 面**爆炸骰**(擲到 20 就再擲一次疊加,細節見
[戰鬥系統](../../guides/combat/)):

```
骰出的總和 + 命中加值 > 目標的 Guard  →  命中
骰出的總和 + 命中加值 ≤ 目標的 Guard  →  落空
```

所以 Guard 就是「要擲到多少才打得中」的門檻。玩家的 Guard 由 DEX 決定
(`clamp(10 + DEX 調整值, 1, 20)`,實際大約落在 7–14),再加上裝備的 Guard。
