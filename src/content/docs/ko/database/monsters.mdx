import FilterTable from '../../../../components/FilterTable.astro';
import { monsters, items, averageRoll, healthForLevel, attackBonusForLevel, damageRollForLevel } from '../../../../lib/gamedata.js';
import { monsterName, itemName, behaviorNames } from '../../../../lib/ko.js';

export const mark = (value, derived) => (derived ? `${value} *` : `${value}`);
export const byLevel = (a, b) => a.level - b.level;

현재 **{monsters.length}종**입니다.

:::caution[`*`가 붙은 값은 유도된 값입니다]
게임 데이터에서 체력, 명중 보너스, 피해 주사위가 **비어 있으면 「덮어쓰지 않음」**을
뜻하고, 서버가 레벨에서 자동으로 계산합니다. 빈칸은 0이 아닙니다.
그래서 여기서는 서버와 같은 공식(`server/src/game/combat.rs`)으로 실제 값을 채우고
`*`로 표시합니다:

- 체력 = `올림(레벨 × 9 ÷ 2)`
- 명중 보너스 = `내림(레벨 ÷ 2)`
- 피해 주사위 = 레벨별 표(아래)
:::

## 전체 몬스터

<FilterTable
  placeholder="몬스터 검색…"
  columns={[
    { key: 'name', label: '이름' },
    { key: 'en', label: '게임 내 이름' },
    { key: 'level', label: '레벨', align: 'right' },

    { key: 'atk', label: '명중 보너스', align: 'right' },
    { key: 'dmg', label: '피해', align: 'center' },

  rows={monsters.slice().sort(byLevel).map((m) => ({
    name: monsterName(m.id),
    en: m.name,
    level: m.level,
    hp: mark(m.healthEffective, m.healthDerived),
    guard: m.guard,
    atk: mark(`+${m.attackBonusEffective}`, m.attackBonusDerived),
    dmg: mark(m.damageRollEffective, m.damageRollDerived),
    avg: averageRoll(m.damageRollEffective)?.toFixed(1) ?? '—',
    behavior: behaviorNames[m.behavior] ?? m.behavior,
  }))}
/>

**행동**은 두 가지뿐입니다. *선제공격*형은 보면 달려들고, *반격*형은 당신이 먼저
때려야 반응합니다. 코볼트와 스토커가 반격형이라 싸우기 싫으면 그냥 지나칠 수 있습니다.

## 이동과 공격 파라미터

도망칠 수 있는지, 공격 사이의 틈을 노릴 수 있는지 판단할 때 보세요.
추적 거리는 쫓아오기 시작하는 거리, 공격 거리는 때릴 수 있는 거리입니다.

플레이어의 기본 이동 속도는 **3.0**(배부르면 ×1.1로 3.3)입니다. 위 표와 비교하면
스토커의 달리기 8.0은 플레이어보다 훨씬 빨라 **도망칠 수 없습니다.**
오크 4.5, 고블린 족장 4.0도 따라옵니다. 코볼트(3.2) 정도만 겨우 떨쳐낼 수 있습니다.

## 출현 위치

던전 몬스터는 층 깊이에 따라, 지상 몬스터는 지정된 구역에 생성됩니다.
가중치가 높을수록 그 심도 범위에서 자주 나옵니다.

<FilterTable
  placeholder="몬스터 검색…"
  columns={[
    { key: 'name', label: '이름' },
    { key: 'depth', label: '던전 심도' },
    { key: 'weight', label: '생성 가중치', align: 'right' },
    { key: 'note', label: '비고' },

  rows={monsters.slice().sort(byLevel).map((m) => ({
    name: monsterName(m.id),

    weight: m.dungeonWeight ?? '—',
    note: m.boss ? '보스, 던전 최하층에 고정 배치' : '',
  }))}
/>

## 드랍

몬스터가 들고 있는 무기가 곧 드랍품입니다 — **보이는 것이 곧 얻을 수 있는 것**이라는
의도적인 설계입니다. 일반 몬스터는 10%, 보스는 100% 확정입니다.

<FilterTable
  placeholder="몬스터나 드랍품 검색…"
  columns={[
    { key: 'name', label: '몬스터' },
    { key: 'drop', label: '드랍 무기' },
    { key: 'chance', label: '드랍률', align: 'right' },
    { key: 'dice', label: '무기 피해', align: 'center' },
  ]}
  rows={monsters.filter((m) => m.weapon).sort(byLevel).map((m) => ({
    name: monsterName(m.id),
    drop: itemName(m.weapon),
    chance: `${Math.round(m.weaponDropChance * 100)}%`,
    dice: items.find((i) => i.id === m.weapon)?.dice ?? '—',
  }))}
/>

오크 워로드는 현재 **철검**(1d8)을 확정으로 주는 유일한 출처입니다 —
철검은 상점에서 팔지 않으므로 이 보스를 잡는 것이 가장 직접적인 경로입니다.

## 레벨별 기본값

비어 있는 항목은 이 표의 값을 씁니다.

/>

## 명중 판정

공격할 때 20면체 주사위를 굴립니다:

```
d20 + 명중 보너스 >  대상의 Guard   →  명중
d20 + 명중 보너스 ≤  대상의 Guard   →  빗나감
```

즉 Guard는 「얼마를 넘겨야 맞는가」의 기준값입니다. 플레이어의 Guard는 DEX로
결정되며(`clamp(10 + DEX 보정, 1, 20)`, 실제로는 대략 7–14), 여기에 장비의 Guard가
더해집니다.
