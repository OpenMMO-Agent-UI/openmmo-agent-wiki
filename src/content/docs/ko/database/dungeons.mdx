import FilterTable from '../../../../components/FilterTable.astro';
import { dungeons, monsters, items, lockedFloors } from '../../../../lib/gamedata.js';
import { dungeonNames, monsterName, itemName } from '../../../../lib/ko.js';

export const dName = (d) => dungeonNames[d.id] ?? d.name;
export const signature = (d) => (d.chestDrops.length ? d.chestDrops.map(itemName).join(', ') : '—');

현재 **{dungeons.length}곳**이 열려 있습니다. 던전은 **시드로 결정**되므로
같은 입구는 들어갈 때마다 구조가 같고, 다시 생성되지 않습니다.

## 보스

<FilterTable
  placeholder="보스 검색…"
  columns={[
    { key: 'name', label: '보스' },
    { key: 'dungeon', label: '던전' },
    { key: 'level', label: '레벨', align: 'right' },
    { key: 'hp', label: '체력', align: 'right' },
    { key: 'guard', label: 'Guard', align: 'right' },
    { key: 'atk', label: '명중 보너스', align: 'right' },
    { key: 'dmg', label: '피해', align: 'center' },
  ]}
  rows={dungeons.filter((d) => d.boss).map((d) => {
    const m = monsters.find((x) => x.id === d.boss);
    return {
      name: monsterName(m.id), dungeon: dName(d),
      level: m.level, hp: m.healthEffective, guard: m.guard,
      atk: `+${m.attackBonusEffective}`, dmg: m.damageRollEffective,
    };
  })}
/>

**오크 워로드**는 격차가 큽니다 — 체력 120은 데이터에 직접 지정된 값이고
(레벨에서 유도되는 45가 아닙니다), Guard 13, 명중 +6, 피해 2d8입니다.
오크 소굴은 10층이니 가죽 세트를 갖추고 내려가는 것을 권합니다.

보스는 죽어도 타이머로 리스폰하지 않습니다 — 해당 층은 그날의 **일몰 던전
리셋**(게임 하루에 한 번)이 올 때까지 죽은 채로 유지되며, 리셋되면 그
던전에 남아 있던 모두가 입구로 순간이동되고 멀리서 포효 소리가 들립니다.
상자를 열 때 더 이상 보스를 잡거나 그 옆에 서 있을 필요가 없습니다 —
아래 「열쇠와 잠긴 층」 참고. 보스가 직접 주던 무기(고블린 검, 철검, 강철
장검)도 이제 상자의 확정 드랍으로 옮겨졌습니다. 아래 표 참고.

## 열쇠와 잠긴 층

모든 던전은 **5층마다** 계단방 출구에 항상 존재하는 잠긴 문이 있어 길을
완전히 막습니다:

<FilterTable
  placeholder="던전 검색…"
  columns={[
    { key: 'name', label: '던전' },
    { key: 'floors', label: '잠긴 층' },
    { key: 'chest', label: '상자를 여는 열쇠' },
  ]}
  rows={dungeons
    .filter((d) => d.keyPrefix)
    .map((d) => {
      const locked = lockedFloors(d);
      return {
        name: dName(d),
        floors: locked.map((f) => `${f}층`).join(', '),
        chest: itemName(`${d.keyPrefix}_${locked[locked.length - 1]}`),
      };
    })}
/>

- **열쇠는 잠긴 문 바로 앞 구간에서 드랍됩니다**(예: 오크 소굴 1–4층은 5층
  열쇠를, 6–9층은 10층 열쇠를 드랍). 그 구간의 몬스터를 처치하면 **5%**
  확률로 해당 열쇠가 드랍되고, 통·상자·잡동사니 상자를 부수면 **1%**입니다.
  이미 그 열쇠를 갖고 있거나 오늘 밤 그 던전 상자를 이미 열었다면 판정을
  건너뜁니다.
- 열쇠는 바닥에 떨어지므로 누구나 주울 수 있습니다. **5층 열쇠로는 10층
  문을 열 수 없습니다.**
- 열쇠를 가진 채로 잠긴 문을 클릭하면 열립니다. **문은 열쇠를 소모하지
  않으므로** 죽거나 나갔다 돌아와도 그대로 쓸 수 있고, 유효기간도 없습니다.
  열린 문은 **30초 뒤 자동으로 닫히며**, 그 사이 열쇠 없는 사람이 따라
  들어갈 수는 있지만 문이 닫히면 어느 쪽에 있든 다시 열 수 없어 계속
  내려가는 수밖에 없습니다.
- **던전에서 가장 깊은 잠긴 문의 열쇠**(오래된 납골당 5층, 오크 소굴 10층,
  오우거 요새 15층)가 상자를 여는 열쇠입니다 — 이걸 들고 최하층까지 가면
  보스를 잡지 않아도 상자를 열 수 있습니다. 여는 순간 그 던전에서
  **소지한 모든 종류의 열쇠**를 한 개씩 소모합니다. 보스는 여전히 상자방을
  지키고 있으므로, 안전하게 열려면 보통 보스를 처리해야 합니다.

## 상자 드랍 규칙

각 던전의 상자 드랍은 세 부분입니다:

1. **확정 드랍(시그니처)** — 던전마다 한두 개씩 고정. 위 표의 「확정 드랍」
   열이며, 세 던전의 보스 무기도 이제 여기 포함됩니다.
2. **개별 굴림** — 해당 티어 풀의 나머지 장비를 각자의 확률로 한 번씩 굴립니다.
3. **하위 티어 보정** — 던전보다 낮은 티어의 장비는 각각 10% 추가 확률.
   상위 던전에서 놓친 부위를 메울 수 있습니다.

골드는 `깊이 × 500 ~ 깊이 × 1500`입니다.

설계 목표는 해당 티어에서 처음 나오는 부위를 모으는 데 **평균 약 5회**입니다.

<FilterTable
  placeholder="장비 검색…"
  columns={[
    { key: 'name', label: '장비' },
    { key: 'en', label: '게임 내 이름' },
    { key: 'tier', label: '상자 티어', align: 'center' },
    { key: 'chance', label: '드랍 확률', align: 'right' },
    { key: 'guard', label: 'Guard', align: 'right' },
    { key: 'cha', label: 'CHA', align: 'right' },
    { key: 'effect', label: '효과' },
  ]}
  rows={items
    .filter((i) => i.chestTier)
    .sort((a, b) => a.chestTier - b.chestTier || (b.guard ?? 0) - (a.guard ?? 0))
    .map((i) => ({
      name: itemName(i.id), en: i.name,
      tier: `T${i.chestTier}`,
      chance: i.chestChance ? `${Math.round(i.chestChance * 100)}%` : '확정 드랍',
      guard: i.guard ?? '—',
      cha: i.chaBonus ? `+${i.chaBonus}` : '—',
      effect: i.hasSustenance ? '포만도 소모 ×0.75' : '—',
    }))}
/>

:::note[상자에서 나오는 무기는 보스 무기 세 개뿐입니다]
무기는 보통 「그 무기를 든 몬스터가 드랍」으로 통일되어 있습니다 — 예외는 고블린 검,
철검, 강철 장검으로, 이제 보스 드랍이 아니라 상자의 확정 드랍입니다. 나머지 무기는
그대로 보이는 것이 곧 얻을 수 있는 것이며, 상자에는 들어 있지 않습니다.
:::

## 던전 안의 다른 것들

- **통과 나무통과 상자를 부술 수 있습니다** — 클릭하면 캐릭터가 다가가 한 번 휘두릅니다.
  보물 상자는 부서지지 않습니다.
- **보물 상자는 여는 것** — 클릭하면 다가가 뚜껑을 열고, 앞에 동전 무더기가 떨어집니다.
  주우면 지갑으로 바로 들어갑니다(1–10 동화). 가방 칸을 차지하지 않습니다.
- **복도에 문이 있습니다** — 복도가 방으로 들어가는 입구에 나무 양문이 있고 클릭해서 엽니다.
- **복도와 방의 벽이 다릅니다** — 방은 중세 석재, 복도는 암벽입니다.
  지금 어디에 있는지 구분하는 데 쓸 수 있습니다.
