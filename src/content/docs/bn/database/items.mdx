import FilterTable from '../../../../components/FilterTable.astro';
import { items, merchants, worldDrops, monsters, formatCoins, sellPrice, averageRoll } from '../../../../lib/gamedata.js';
import { itemName, itemDescriptions, slotNames, materialNames, categoryNames, monsterName } from '../../../../lib/bn.js';

export const rica = merchants.find((m) => m.id === 'rica');
export const dropChance = new Map(worldDrops.map((d) => [d.id, d.chance]));
export const monsterWeapons = monsters.filter((m) => m.weapon);

export const otherMerchants = merchants.filter((m) => m.id !== 'rica');

export const sources = (id) => {
  const out = [];
  if (rica.catalog.includes(id)) out.push('দোকান');
  else {
    const seller = otherMerchants.find((m) => m.catalog.includes(id));
    if (seller) out.push(`Shop (${seller.npcName})`);
  }
  if (dropChance.has(id)) out.push(`World drop ${(dropChance.get(id) * 100).toFixed(1)}%`);
  for (const m of monsterWeapons.filter((m) => m.weapon === id && m.weaponDropChance > 0)) {
    out.push(`${monsterName(m.id)} ${Math.round(m.weaponDropChance * 100)}%`);
  }
  const it = items.find((i) => i.id === id);
  if (it?.chestTier) out.push(`Chest T${it.chestTier}${it.chestChance ? ` ${Math.round(it.chestChance * 100)}%` : ' guaranteed'}`);
  if (it?.grillsInto) out.push('মাছ ধরা');
  return out.length ? out.join(', ') : '—';
};

export const byId = (a, b) => a.id.localeCompare(b.id);

All **{items.length} items**, read straight from the game's `data-src/items.csv`
so they match what the server loads.

:::note[Reading prices]
The game stores one copper integer and converts on display:
**1 gold (g) = 100 silver (s) = 10,000 copper (c)**.
Items with no price (the "পরা" ones) are **refused by merchants** — that's the
design that stops starting gear being cashed out. Those three "পরা" starting
items are also marked **untradeable to other players** — see
[Player trading](../../guides/trade/).

**Sell price** is what merchant Rica pays, fixed at
**{rica.sellRatePercent}%** of base. The spread means buying low and selling
high doesn't work.
:::

## Weapons

Damage is dice notation — `1d8` is one eight-sided die. The **average** column
is there so you can compare directly.

<FilterTable
  placeholder="অস্ত্র খুঁজুন…"
  columns={[
    { key: 'name', label: 'নাম' },
    { key: 'dice', label: 'ক্ষতি', align: 'center' },
    { key: 'avg', label: 'গড়', align: 'right' },
    { key: 'weight', label: 'ভার', align: 'right' },
    { key: 'price', label: 'দোকান', align: 'right' },
    { key: 'sell', label: 'বিক্রি হয়', align: 'right' },
    { key: 'source', label: 'কোথায় পাবেন' },
  ]}
  rows={items
    .filter((i) => i.category === 'weapon')
    .sort((a, b) => (averageRoll(b.dice) ?? 0) - (averageRoll(a.dice) ?? 0))
    .map((i) => ({
      name: i.name,
      dice: i.dice ?? '—',
      avg: averageRoll(i.dice)?.toFixed(1) ?? '—',
      weight: `${i.weight} kg`,
      price: formatCoins(i.basePrice),
      sell: formatCoins(sellPrice(i.basePrice, rica.sellRatePercent)),
      source: sources(i.id),
    }))}
/>

:::note[The Great Sword's "সিন্দুক T4 নিশ্চিত" isn't obtainable yet]
The Great Sword's data is live, but the three dungeons' chests only go up to
T3 — nothing currently reaches T4. It's data parked ahead of a dungeon that
hasn't shipped yet, so that chest tag is a placeholder for now. See
[Dungeons](../dungeons/).
:::

## Armor

**Guard** is the number an attacker has to beat to hit you — higher is better.
Worn Guard adds up across slots.

The Gold Ring is a ring of adornment: CHA +1, which widens your haggling price
band by ±2 percentage points. It drops from tier-3 dungeon chests (10% roll) —
every ring already in circulation from the old chest rules gets the effect
retroactively. The Silver Necklace slows satiation drain to 0.75x while
equipped (sprinting included), and drops from tier-2 dungeon chests (10%
roll) — every necklace already in circulation from the old chest rules gets
the effect retroactively too.

## Catches

Raw fish are edible (fixed 40 nutrition) but **carry a food poisoning risk**.
Grilling removes it. See [Fishing](../fishing/).

<FilterTable
  placeholder="ধরা মাছ খুঁজুন…"

      grill: i.grillsInto ? itemName(i.grillsInto) : 'ঝলসানো যায় না',
    }))}
